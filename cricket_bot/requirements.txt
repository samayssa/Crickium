# same `import pyrogram` namespace (every file in this project already
# does `from pyrogram import ...` / `from pyrogram.types import ...`
# and needs ZERO changes), but adds support for the newer Bot API
# features including button styles. If pyrogram is already installed,
# remove it first so the two don't conflict:
#   pip uninstall pyrogram -y
#   pip install -r requirements.txt
